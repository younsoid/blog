```python
# add_sid 매칭 파일(xml_sid)에 대해서만 정제 수행
BASE_DIR = r"\\10.32.91.11\core\03. [로슈]정밀 의료 생태계 구축 사업\04. FMI"

bio_rows, vaf_rows, failures = [], [], []
for fname in xml_sid["file_name"]:
    path = os.path.join(BASE_DIR, fname)
    try:
        bio, vafs = parse_report(path)
        bio_rows.append(bio)
        vaf_rows.extend(vafs)
    except Exception as e:
        failures.append((fname, repr(e)))

print(f"대상 {len(xml_sid)}개 → 성공 {len(bio_rows)} / 실패 {len(failures)}")
print(f"BIO {len(bio_rows)}행, VAF {len(vaf_rows)}행")
if failures:
    for fn, err in failures:
        print("  실패:", fn, err)
```

### 3. 결과 확인

먼저 시트의 각 컬럼들을 어디서 가져오는지를 정리해보면 아래와 같다.

BIO 시트
- 확인일 - 오늘 날짜로 통일
- MEDICAL RECORD - FinalReport > PMI > MRN
- FIRST NAME - FinalReport > PMI > FirstName
- LAST NAME - FinalReport > PMI > LastName
- DISEASE - FinalReport > PMI > SubmittedDiagnosis
- DATE OF BIRTH - FinalReport > PMI > DOB          (YYYY-MM-DD → YYYYMMDD)
- SEX - FinalReport > PMI > Gender
- SPECIMEN SITE - FinalReport > PMI > SpecSite
- DATE OF COLLECTION - FinalReport > PMI > CollDate     (→ YYYYMMDD)
- SPECIMEN RECEIVED - FinalReport > PMI > ReceivedDate (→ YYYYMMDD)
- SPECIMEN TYPE - FinalReport > Sample > SpecFormat
- MedianCoverageValue - FinalReport > reportProperties > reportProperty[key="MedianCoverageValue"] > value
- TumorMutationBurdenScore - FinalReport > reportProperties > reportProperty[key="TumorMutationBurdenScore"] > value
- MicrosatelliteStatusScore - FinalReport > reportProperties > reportProperty[key="MicrosatelliteStatusScore"] > value
- LossOfHeterozygosityScore - FinalReport > reportProperties > reportProperty[key="LossOfHeterozygosityScore"] > value
- BloodTumorMutationalBurdenScore  - FinalReport > reportProperties > reportProperty[key="BloodTumorMutationalBurdenScore"] > value
- TumorFractionScore - FinalReport > reportProperties > reportProperty[key="TumorFractionScore"] > value
- HomologousRecombinationDeficiencyScore - FinalReport > reportProperties > reportProperty[key="HomologousRecombinationDeficiencyScore"] > value
- ComprehensiveTumorFractionScore - FinalReport > reportProperties > reportProperty[key="ComprehensiveTumorFractionScore"] > value

VAF 시트
- 확인일 - 오늘 날짜로 통일
- MEDICAL RECORD - FinalReport > PMI > MRN
- FIRST NAME - FinalReport > PMI > FirstName
- LAST NAME - FinalReport > PMI > LastName
- GENE - variant-report > short-variants > short-variant > @gene
- ALTERATION - variant-report > short-variants > short-variant > @protein-effect
- TRANSCRIP_ID - variant-report > short-variants > short-variant > @transcript
- CODING_SEQUENCE_EFFECT - variant-report > short-variants > short-variant > @cds-effect
- VARIANT_CHROMOSOMAL_POSITION - variant-report > short-variants > short-variant > @position
- VAF - variant-report > short-variants > short-variant > @percent-reads
